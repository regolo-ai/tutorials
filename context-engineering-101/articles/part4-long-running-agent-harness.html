Ask a coding agent to build or refactor a multi-module service over several hours, and you will quickly hit two failure modes that no single system prompt can fix. First, the agent tries to one-shot the entire specification in a single marathon loop until its context window fills with half-finished edits. Second, when a subsequent session resumes on the repository, it glances at the existing files, assumes the hard cases work, and declares victory prematurely.

Compaction alone is not enough when a task spans dozens of features across multiple context windows. As Anthropic Engineering and LangChain demonstrated in their long-horizon harness research, reliable multi-hour execution requires treating the filesystem and Git as the durable source of truth and separating the agent that writes code from the agent that grades it.

#!/usr/bin/env python3
"""piece4_long_running_harness.py: Initializer + Context-Reset Worker + Independent Evaluator."""
import json, os, ssl, urllib.request
from pathlib import Path

REGOLO_BASE_URL = os.environ.get("REGOLO_BASE_URL", "https://api.regolo.ai/v1")
WORKSPACE_DIR = Path(".harness/long_running_demo")
_SSL_CTX = ssl._create_unverified_context()

def _chat(model: str, system: str, user: str, api_key: str, max_tokens: int = 300) -> dict:
    req = urllib.request.Request(
        f"{REGOLO_BASE_URL}/chat/completions",
        data=json.dumps({
            "model": model,
            "messages": [{"role": "system", "content": system}, {"role": "user", "content": user}],
            "max_tokens": max_tokens,
            "temperature": 0.0,
            "chat_template_kwargs": {"enable_thinking": False},
        }).encode("utf-8"),
        headers={"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"},
        method="POST",
    )
    with urllib.request.urlopen(req, timeout=60, context=_SSL_CTX) as resp:
        return json.loads(resp.read().decode("utf-8"))

def initialize_harness(spec: str) -> Path:
    WORKSPACE_DIR.mkdir(parents=True, exist_ok=True)
    features = [
        {"id": "feat_1", "title": "Parse EU VAT invoice amount", "passes": False},
        {"id": "feat_2", "title": "Reject negative invoice totals with ValueError", "passes": False},
    ]
    fl_path = WORKSPACE_DIR / "feature_list.json"
    fl_path.write_text(json.dumps(features, indent=2), encoding="utf-8")
    (WORKSPACE_DIR / "progress.txt").write_text("# Session Progress Log\n- Initialized feature_list.json\n", encoding="utf-8")
    return fl_path

"""regolo_harness_mcp.py: Unified 4-tool MCP stdio server for OpenCode, Claude Code, and Codex CLI."""
from __future__ import annotations

from pathlib import Path
from piece1_progressive_skills import ProgressiveSkillRegistry, format_tool_response

TOOLS = [
    {
        "name": "load_skill_on_demand",
        "description": "Part 1: Load instructions for a named skill only when needed (progressive disclosure).",
        "inputSchema": {
            "type": "object",
            "properties": {
                "skill_name": {"type": "string"},

        "inputSchema": {

    },

        "description": "Part 4: Initialize .harness/long_running_demo/feature_list.json and progress.txt for multi-session execution.",
        "inputSchema": {
            "type": "object",
            "properties": {"spec": {"type": "string"}},
            "required": ["spec"]
        }
    }
]

def handle_rpc(req: dict) -> dict | None:
    method, req_id = req.get("method"), req.get("id")
    if method == "initialize":
        return {"jsonrpc": "2.0", "id": req_id, "result": {"protocolVersion": "2024-11-05", "capabilities": {"tools": {}}, "serverInfo": {"name": "regolo-context-harness", "version": "1.0.0"}}}
    if method == "tools/list":
        return {"jsonrpc": "2.0", "id": req_id, "result": {"tools": TOOLS}}
    if method == "tools/call":
        params = req.get("params", {})
        name, args = params.get("name"), params.get("arguments", {})
        api_key = os.environ.get("REGOLO_API_KEY", "")
        if name == "load_skill_on_demand":
            reg = ProgressiveSkillRegistry(Path(os.environ.get("REGOLO_SKILLS_DIR", "~/.claude/skills")).expanduser())
            text = reg.load_skill(args["skill_name"])
            out = "\n".join(text.splitlines()[:40]) if args.get("response_format", "concise") == "concise" else text
        elif name == "offload_and_delegate":
            off = offload_tool_output(args["tool_name"], args["raw_output"])
            sub = delegate_chunk_analyst(off["saved_path"], args["question"], api_key) if args.get("question") and off.get("saved_path") and api_key else None
            out = json.dumps({"offload": off, "subagent": sub}, indent=2)
        elif name == "reversible_compact_history":
            out = json.dumps(reversible_compact(args["messages"], args["session_id"], api_key), indent=2)
        elif name == "init_feature_ledger":
            out = f"Initialized {initialize_harness(args['spec'])}"
        else:
            out = f"Unknown tool: {name}"
        return {"jsonrpc": "2.0", "id": req_id, "result": {"content": [{"type": "text", "text": out}]}}
    return None

if __name__ == "__main__":
    if "--self-test" in sys.argv:
        r1 = handle_rpc({"jsonrpc": "2.0", "id": 1, "method": "initialize"})
        r2 = handle_rpc({"jsonrpc": "2.0", "id": 2, "method": "tools/list"})
        assert r1 and r2 and len(r2["result"]["tools"]) == 4
        print("MCP JSON-RPC stdio self-test PASSED: 4 tools registered:", [t["name"] for t in r2["result"]["tools"]])
        sys.exit(0)
    for line in sys.stdin:
        if line.strip():
            resp = handle_rpc(json.loads(line))
            if resp is not None:
                sys.stdout.write(json.dumps(resp) + "\n")
                sys.stdout.flush()

## Long-running feature ledger rules (AGENTS.md / CLAUDE.md)
1. At the start of a multi-feature specification, call `init_feature_ledger(spec="...")` to create `.harness/long_running_demo/feature_list.json` (with all features set to `"passes": false`) and `progress.txt`.

4. Before resetting or closing the session, append a 1-line checkpoint to `progress.txt` so the next agent session starts from a clean window.

Frequently asked questions

Why do long-running coding agents fail when asked to build an entire feature set in one session?
Agents suffer from two opposite failure modes on multi-hour tasks: they either try to one-shot the entire application and run out of clean context halfway through, or they see partial progress and prematurely declare the project finished.

What is the Initializer, Worker, and Evaluator harness pattern?
An Initializer agent decomposes the specification into a structured feature_list.json file where every item starts at passes=false. A Worker agent implements one feature at a time in a clean context window, and an independent Evaluator agent runs tests and audits the output before marking passes=true.

When is a hard context reset better than conversation compaction?
When moving from one completed feature to the next in a multi-hour build, carrying even a summarized chat transcript can introduce stale assumptions. A context reset starts the Worker with a fresh window containing only feature_list.json, progress.txt, and git status.

Why use JSON instead of Markdown for feature_list.json?
As Anthropic Engineering observed in long-running harness benchmarks, coding models are much less likely to accidentally delete or rephrase acceptance criteria inside a strict JSON schema than inside a free-form Markdown checklist.
