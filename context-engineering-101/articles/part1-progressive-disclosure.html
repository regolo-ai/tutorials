Two harness primitives: progressive skill menus and concise tool responses

As highlighted in LangChain’s Deep Agents guide and Anthropic’s tool engineering research, keeping irrelevant tokens out of the window starts with two structural changes in your harness:

  1. Progressive disclosure for skills: Instead of injecting the full body of every SKILL.md file into the system prompt, scan your skills folder at startup and extract only the skill name and its first-sentence description. Ten skills occupy roughly three hundred tokens instead of nearly forty thousand. When the agent encounters a task that matches a skill, it calls load_skill_on_demand(skill_name) to pull the full instructions just in time.
  2. A response_format toggle on every tool: Add an enum parameter ("concise" vs "detailed") to your custom tools. Default to "concise" so daily searches return human-readable titles, statuses, and summaries without cryptographic hashes or internal IDs. When the agent needs an ID to trigger a mutation call, it requests response_format="detailed" for that single record.

Build, test, and validate Piece 1 in Python

Save the script below as piece1_progressive_skills.py (also available in our context-engineering-101 GitHub kit). Notice one practical detail when calling reasoning-capable models like qwen3.8-27b on routine harness turns: we pass "chat_template_kwargs": {"enable_thinking": False} so the model answers directly in five seconds instead of spending extra budget on an internal chain-of-thought scratchpad:

#!/usr/bin/env python3
"""piece1_progressive_skills.py: Progressive skill menu + concise tool formatter."""
import json, os, re, ssl, time, urllib.request
from pathlib import Path

REGOLO_BASE_URL = os.environ.get("REGOLO_BASE_URL", "https://api.regolo.ai/v1")
_SSL_CTX = ssl._create_unverified_context()

def estimate_tokens(text: str) -> int:
    return max(1, len(text) // 4)

class ProgressiveSkillRegistry:
    def __init__(self, skills_dir: Path, max_skills: int = 10):
        self.skills_dir = Path(skills_dir).expanduser()
        self._index = {}
        for skill_md in sorted(self.skills_dir.glob("*/SKILL.md"))[:max_skills]:
            raw = skill_md.read_text(encoding="utf-8", errors="ignore")
            m = re.search(r"^description:\s*(.+)$", raw, re.MULTILINE)
            desc = m.group(1).strip().strip("'"") if m else "Specialized workflow."
            summary = re.split(r"(?<=[.!?])\s+", desc)[0][:110]
            self._index[skill_md.parent.name] = {"path": str(skill_md), "summary": summary}

    def compact_menu(self) -> str:
        lines = ["Available on-demand skills (call load_skill(name) only when needed):"]
        return "
".join(lines + [f"- {k}: {v['summary']}" for k, v in self._index.items()])

    def load_skill(self, name: str) -> str:
        if name not in self._index:
            return f"Unknown skill '{name}'. Available: {', '.join(self._index)}"
        return Path(self._index[name]["path"]).read_text(encoding="utf-8", errors="ignore")

def format_tool_response(records: list[dict], response_format: str = "concise") -> str:
    if response_format == "detailed":
        return json.dumps(records, indent=2)
    keep = ("title", "status", "owner", "summary", "line", "file")
    return json.dumps([{k: v for k, v in r.items() if k in keep} for r in records], separators=(",", ":"))

Step 1: Create the MCP server file (regolo_harness_mcp.py)

In the same folder where you saved piece1_progressive_skills.py, create a new file named regolo_harness_mcp.py and paste the code below. It uses only the Python standard library to speak JSON-RPC 2.0 over stdin/stdout, importing ProgressiveSkillRegistry and format_tool_response from piece1_progressive_skills.py and exposing them as two callable MCP tools (load_skill_on_demand and format_concise_tool_output):

#!/usr/bin/env python3
"""regolo_harness_mcp.py: MCP stdio server for OpenCode, Claude Code, and Codex CLI."""
from __future__ import annotations
import json, os, sys
from pathlib import Path
from piece1_progressive_skills import ProgressiveSkillRegistry, format_tool_response

SKILLS_DIR = Path(os.environ.get("REGOLO_SKILLS_DIR", "~/.claude/skills")).expanduser()

TOOLS = [
    {
        "name": "load_skill_on_demand",
        "description": "Load instructions for a named skill only when the current task requires it (progressive disclosure).",
        "inputSchema": {
            "type": "object",
            "properties": {
                "skill_name": {"type": "string", "description": "Exact skill folder name from the AGENTS.md menu."},
                "response_format": {"type": "string", "enum": ["concise", "detailed"], "default": "concise"}
            },
            "required": ["skill_name"]
        }
    },
    {
        "name": "format_concise_tool_output",
        "description": "Strip low-signal metadata (UUIDs, SHA hashes, raw stack traces) from a JSON tool payload.",
        "inputSchema": {
            "type": "object",
            "properties": {
                "raw_json": {"type": "string", "description": "Raw JSON string returned by an external tool."},
                "response_format": {"type": "string", "enum": ["concise", "detailed"], "default": "concise"}
            },
            "required": ["raw_json"]
        }
    }
]

def handle_rpc(req: dict) -> dict | None:
    method, req_id = req.get("method"), req.get("id")
    if method == "initialize":
        return {
            "jsonrpc": "2.0",
            "id": req_id,
            "result": {
                "protocolVersion": "2024-11-05",
                "capabilities": {"tools": {}},
                "serverInfo": {"name": "regolo-context-harness", "version": "1.0.0"}
            }
        }
    if method == "tools/list":
        return {"jsonrpc": "2.0", "id": req_id, "result": {"tools": TOOLS}}
    if method == "tools/call":
        params = req.get("params", {})
        name, args = params.get("name"), params.get("arguments", {})
        if name == "load_skill_on_demand":
            registry = ProgressiveSkillRegistry(SKILLS_DIR)
            text = registry.load_skill(args["skill_name"])
            if args.get("response_format", "concise") == "concise":
                text = "\n".join(text.splitlines()[:40])
        elif name == "format_concise_tool_output":
            payload = json.loads(args["raw_json"])
            text = format_tool_response(payload, response_format=args.get("response_format", "concise"))
        else:
            text = f"Unknown tool: {name}"
        return {"jsonrpc": "2.0", "id": req_id, "result": {"content": [{"type": "text", "text": text}]}}
    return None

if __name__ == "__main__":
    if "--self-test" in sys.argv:
        init_ok = handle_rpc({"jsonrpc": "2.0", "id": 1, "method": "initialize"})
        list_ok = handle_rpc({"jsonrpc": "2.0", "id": 2, "method": "tools/list"})
        assert init_ok and list_ok and len(list_ok["result"]["tools"]) == 2
        print("MCP stdio handshake PASSED:", [t["name"] for t in list_ok["result"]["tools"]])
        sys.exit(0)
    for line in sys.stdin:
        if line.strip():
            resp = handle_rpc(json.loads(line))
            if resp is not None:
                sys.stdout.write(json.dumps(resp) + "\n")
                sys.stdout.flush()

Frequently asked questions

What is the difference between prompt engineering and context engineering for AI agents?
Prompt engineering focuses on phrasing a single instruction, while context engineering manages the entire set of tokens entering and leaving the model window across dozens of turns, including system instructions, skill files, MCP tool schemas, and tool outputs.

Why do agents lose accuracy even when the model supports a 262K or 1M context window?
Every additional block of irrelevant instructions or raw UUIDs dilutes the attention budget of the transformer, a phenomenon known as context rot. Keeping the startup prefix lean preserves reasoning precision on later turns.

How does progressive disclosure work in OpenCode, Claude Code, and Codex?
Instead of loading full markdown guides for every skill at startup, the harness injects only a one-line name and summary per skill into the system prompt and exposes an on-demand tool so the model reads the full skill file only when a task requires it.

Why should I pass enable_thinking=False on routine harness turns with Qwen3.8-27B?
Reasoning models generate an internal scratchpad by default. Disabling thinking on deterministic formatting or routing calls avoids unnecessary latency and saves output tokens for steps that actually require deep planning.

Which models on the Regolo API work best with progressive skill disclosure?
